def sha256(path) -> str:
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load_module(path, name):
    """Compile a repository script into an in-memory module without running main()."""
    source = Path(path).read_text(encoding="utf-8-sig")
    module = types.ModuleType(name)
    module.__file__ = str(path)
    sys.modules[name] = module            # dataclasses need the module registered
    exec(compile(source, str(path), "exec"), module.__dict__)
    return module

# ----------------------------------------------------------------------------------
# Reference 1: rigorous Arb balls, definitional formulas (independent of the identity
# A = B exp(-d) that the code relies on).
# ----------------------------------------------------------------------------------
def arb_reference(xp: float, xa: float, xb: float, target_rel=1e-30, start_prec=256, max_prec=8192):
    prec = start_prec
    while True:
        flint.ctx.prec = prec
        X = [flint.arb(float(v)) for v in (xp, xa, xb)]        # exact conversion
        n = [1 / x.expm1() for x in X]
        A = n[0] * (1 + n[1]) * (1 + n[2])
        B = (1 + n[0]) * n[1] * n[2]
        d = X[0] - X[1] - X[2]                                   # exact at this precision
        exact_zero = d.is_zero()
        if exact_zero:
            F = flint.arb(0)
            L = A
        else:
            F = A - B
            L = F / (A.log() - B.log())
        good = True
        for ball in ([L] if exact_zero else [F, L]):
            if not ball.is_finite():
                good = False
                break
            if ball.rad() != 0:
                rel = ball.rad() / abs(ball.mid())
                if not (rel < target_rel):
                    good = False
        if good or prec >= max_prec:
            return {"F": F, "L": L, "d_exact_zero": bool(exact_zero), "prec_bits": prec,
                    "A": A, "B": B}
        prec *= 2


def arb_to_mpf(ball, dps=60):
    """Convert an Arb ball midpoint to an mpmath mpf (exact binary midpoint)."""
    with mp.workdps(dps):
        s = ball.mid().str(dps + 5, radius=False)
        return mp.mpf(s)


# ----------------------------------------------------------------------------------
# Reference 2: mpmath identity route with exact affinity and analytic derivatives.
# ----------------------------------------------------------------------------------
def mp_reference(xp: float, xa: float, xb: float, dps=60, derivatives=True):
    with mp.workdps(dps):
        X = [mp.mpf(float(v)) for v in (xp, xa, xb)]            # exact (53-bit) conversion
        # exact affinity: dps=60 (~200 bits) holds the sum of three doubles in [1e-12,1e3]
        # exactly; check by a second evaluation at higher precision.
        d = X[0] - X[1] - X[2]
    with mp.workdps(dps + 400):
        d_check = mp.mpf(float(xp)) - mp.mpf(float(xa)) - mp.mpf(float(xb))
    if d_check != d:
        raise RuntimeError("affinity not exact at working precision")
    # precision boost when |d| is tiny (needed only for derivative of g(d))
    extra = 0
    if d != 0:
        extra = int(max(0, -float(mp.log10(abs(d))))) * 2 + 10
    with mp.workdps(dps + extra):
        n = [1 / mp.expm1(x) for x in X]
        B = (1 + n[0]) * n[1] * n[2]
        A = n[0] * (1 + n[1]) * (1 + n[2])
        em1 = mp.expm1(-d)
        F = B * em1
        if d == 0:
            g = mp.mpf(1)
            gp = mp.mpf(-0.5)
        else:
            g = -em1 / d
            gp = (mp.exp(-d) * (1 + d) - 1) / d**2
        L = B * g
        out = {"F": +F, "L": +L, "A": +A, "B": +B, "d": +d, "logA": mp.log(A), "logB": mp.log(B)}
        if derivatives:
            dlogB = [-n[0], -(1 + n[1]), -(1 + n[2])]
            dd = [1, -1, -1]
            e = mp.exp(-d)
            dF = [B * dlogB[j] * (e - 1) - B * e * dd[j] for j in range(3)]
            dL = [B * dlogB[j] * g + B * gp * dd[j] for j in range(3)]
            out["dF"] = [+v for v in dF]
            out["dL"] = [+v for v in dL]
            out["condF_abs"] = mp.fsum(abs(X[j] * dF[j]) for j in range(3))
            out["condL_abs"] = mp.fsum(abs(X[j] * dL[j]) for j in range(3))
    return out


def mp_definitional_derivatives(xp, xa, xb, dps=220):
    """Second, definitional route for the Jacobian: dF = A dlogA - B dlogB and the
    quotient rule for Lambda = F/(logA-logB); high precision absorbs the cancellation."""
    with mp.workdps(dps):
        X = [mp.mpf(float(v)) for v in (xp, xa, xb)]
        n = [1 / mp.expm1(x) for x in X]
        A = n[0] * (1 + n[1]) * (1 + n[2])
        B = (1 + n[0]) * n[1] * n[2]
        dlogA = [-(1 + n[0]), -n[1], -n[2]]
        dlogB = [-n[0], -(1 + n[1]), -(1 + n[2])]
        F = A - B
        dF = [A * dlogA[j] - B * dlogB[j] for j in range(3)]
        Lg = mp.log(A) - mp.log(B)
        if Lg == 0:
            return {"dF": dF, "dL": None}
        Lam = F / Lg
        dL = [(dF[j] - Lam * (dlogA[j] - dlogB[j])) / Lg for j in range(3)]
        return {"dF": dF, "dL": dL}


def classify_and_error(y: float, ref, dps=60):
    """Error of a float result y against a high-precision reference value (mpf)."""
    if not np.isfinite(y):
        return {"regime": "nonfinite_output", "finite": False}
    with mp.workdps(dps):
        r = mp.mpf(ref)
        ar = abs(r)
        diff = abs(mp.mpf(float(y)) - r)
        if r == 0:
            return {"regime": "exact_zero", "finite": True, "abs_error": float(diff),
                    "is_exact_zero": bool(y == 0.0)}
        if ar >= TINY:
            return {"regime": "normal", "finite": True, "rel_error": float(diff / ar)}
        if ar >= SUB / 2:
            return {"regime": "subnormal", "finite": True, "abs_error_in_subnormal_ulps": float(diff / SUB),
                    "rel_error": float(diff / ar)}
        return {"regime": "below_subnormal", "finite": True, "abs_error_in_subnormal_ulps": float(diff / SUB),
                "output": float(y)}
